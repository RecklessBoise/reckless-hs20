cmake_minimum_required(VERSION 3.22)

project(RecklessHS20 VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")

if(NOT CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "Universal binary" FORCE)
endif()

option(HS20_COPY_PLUGIN_AFTER_BUILD "Install the plug-ins into ~/Library/Audio/Plug-Ins after building" OFF)

add_subdirectory(JUCE)

# ---------------------------------------------------------------------------
# Factory presets are generated by tools/generate_presets.py and embedded.
# ---------------------------------------------------------------------------
juce_add_binary_data(HS20Data
    HEADER_NAME HS20BinaryData.h
    NAMESPACE HS20BinaryData
    SOURCES resources/factory_presets.json)

set(HS20_CORE_SOURCES
    Source/Parameters.cpp
    Source/ParameterBridge.cpp
    Source/dsp/SynthEngine.cpp
    Source/presets/PresetManager.cpp)

set(HS20_UI_SOURCES
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
    Source/ui/HS20LookAndFeel.cpp
    Source/ui/PanelTexture.cpp
    Source/ui/PatchBay.cpp
    Source/ui/PresetBar.cpp
    Source/ui/SynthPanel.cpp)

juce_add_plugin(RecklessHS20
    COMPANY_NAME "Reckless"
    COMPANY_WEBSITE "https://github.com/RecklessBoise/reckless-hs20"
    BUNDLE_ID com.reckless.hs20
    PLUGIN_MANUFACTURER_CODE Rckl
    PLUGIN_CODE Hs20
    PRODUCT_NAME "Reckless HS20"
    DESCRIPTION "Monophonic semi-modular analog-style synthesizer"
    FORMATS VST3 AU Standalone
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    VST3_CATEGORIES Instrument Synth
    AU_MAIN_TYPE kAudioUnitType_MusicDevice
    COPY_PLUGIN_AFTER_BUILD ${HS20_COPY_PLUGIN_AFTER_BUILD})

target_sources(RecklessHS20 PRIVATE ${HS20_CORE_SOURCES} ${HS20_UI_SOURCES})
target_include_directories(RecklessHS20 PRIVATE Source)

target_compile_definitions(RecklessHS20 PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0)

target_link_libraries(RecklessHS20
    PRIVATE
        HS20Data
        juce::juce_audio_utils
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags)

# ---------------------------------------------------------------------------
# Headless test runner: DSP sanity, preset loading, favourites persistence.
# ---------------------------------------------------------------------------
juce_add_console_app(HS20Tests PRODUCT_NAME "HS20Tests")

target_sources(HS20Tests PRIVATE tests/TestMain.cpp ${HS20_CORE_SOURCES})
target_include_directories(HS20Tests PRIVATE Source)
target_compile_definitions(HS20Tests PRIVATE
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_UNIT_TESTS=1)

target_link_libraries(HS20Tests
    PRIVATE
        HS20Data
        juce::juce_audio_basics
        juce::juce_audio_processors
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags)

# ---------------------------------------------------------------------------
# Developer tool: renders the editor to a PNG (used for the README screenshot).
# ---------------------------------------------------------------------------
juce_add_gui_app(HS20Snapshot PRODUCT_NAME "HS20Snapshot")
target_sources(HS20Snapshot PRIVATE tools/Snapshot.cpp ${HS20_CORE_SOURCES} ${HS20_UI_SOURCES})
target_include_directories(HS20Snapshot PRIVATE Source)
target_compile_definitions(HS20Snapshot PRIVATE
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JucePlugin_Name="Reckless HS20"
    JucePlugin_Build_Standalone=0)
target_link_libraries(HS20Snapshot
    PRIVATE
        HS20Data
        juce::juce_audio_utils
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags)
